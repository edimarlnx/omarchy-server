# This file is the template the whole system boots from. The ISO installer
# copies it to the ESP as /boot/limine.conf (_write_limine_defaults reads
# /usr/share/omarchy/default/limine/limine.conf out of the target), and
# `omarchy-refresh-limine` copies it again on demand. limine-snapper-sync and
# limine-entry-tool only add and remove entries below, so everything above the
# first entry survives every regeneration -- which is why the branding belongs
# here and not in a one-off edit of /boot/limine.conf.
#
# Measured on the desktop reference install: the loader accounted for 6.4 s of a
# 12.2 s boot, because the commented-out `timeout: 3` leaves limine's own 5 s
# default in force. Two seconds is the compromise: long enough that the branded
# menu is on screen and reachable without holding a key, short enough that an
# unattended reboot is not waiting on a human.
timeout: 2
default_entry: 2
interface_branding: Omarchy Server
interface_branding_color: 9ece6a
interface_help_color: 9ece6a
interface_help_color_bright: 9ece6a
hash_mismatch_panic: no

# The Omarchy mark on the Tokyo Night background, installed on the ESP beside
# this file by install/server/limine-branding-server.sh. `boot():` is the
# partition the config itself was read from, so the path holds no matter which
# disk the machine boots. Limine skips a missing wallpaper silently instead of
# panicking, so a machine that never ran that script still boots.
wallpaper: boot():/limine-wallpaper.png
# `stretched`, the default, and not `centered`: a centered image larger than the
# framebuffer is cropped, and one smaller than it leaves a border. The artwork
# is a wordmark on a flat field, so scaling it to whatever mode the firmware
# hands over costs nothing and always fills the screen.
wallpaper_style: stretched

# TTRRGGBB, where TT is transparency and 00 means OPAQUE. This is the one that
# is easy to get wrong: a six-digit value is read as TT=00, so `term_background:
# 1a1b26` paints an opaque panel over the wallpaper and the image never shows.
# 80 lets the wallpaper through at half strength, which keeps the menu text as
# readable as it is on a flat background.
term_background: 80000000
# Fills whatever the wallpaper does not cover, and the whole screen when the
# image is missing. Same colour as the artwork's own background.
backdrop: 1a1b26

# Terminal colors (Tokyo Night palette)
term_palette: 15161e;f7768e;9ece6a;e0af68;7aa2f7;bb9af7;7dcfff;a9b1d6
term_palette_bright: 414868;f7768e;9ece6a;e0af68;7aa2f7;bb9af7;7dcfff;c0caf5

# Text colors
term_foreground: c0caf5
term_foreground_bright: c0caf5
term_background_bright: 24283b

TARGET_OS_NAME="Omarchy Server"

# Server cmdline. Differences from the desktop defaults:
#   - no `quiet splash loglevel=0 systemd.show_status=false
#     rd.udev.log_level=0 vt.global_cursor_default=0`: a headless machine has
#     nothing to hide the boot behind, and hiding it is exactly what makes a
#     failed boot undebuggable.
#   - no `initramfs_async=0`: that workaround exists only because plymouthd
#     races the asynchronous initramfs unpack, and plymouth is not installed
#     in this profile.
#   - console=ttyS0,115200 console=tty0: the last console= gets /dev/console,
#     so tty0 stays primary and the serial port receives a copy. Paired with
#     serial-getty@ttyS0.service from install/server/enable-services-server.sh.
#   - no resume=/resume_offset=: the server profile creates no swapfile and
#     skips the configure_hibernation phase; zram is the only swap
#    .
KERNEL_CMDLINE[default]+=" console=ttyS0,115200 console=tty0"

# Server hook set. Differences from the desktop omarchy_hooks.conf:
#   - no `plymouth` hook: the package is not installed in this profile, and an
#     absent hook makes mkinitcpio fail outright. On an
#     encrypted install the LUKS prompt becomes cryptsetup's plain text
#     prompt, which is what a serial console can actually render.
#   - no NVIDIA/kms conditional block: install/hardware/nvidia.sh is not part
#     of the server profile, so MODULES never contains nvidia_drm.
#   - `consolefont` and `keymap` are kept: they still apply to the text LUKS
#     prompt on the local tty.
HOOKS=(base udev keyboard autodetect microcode modconf kms keymap consolefont block encrypt filesystems fsck btrfs-overlayfs)

# Bundle vconsole.conf so the LUKS prompt uses the configured keyboard layout,
# but only when that layout types Latin letters. Passphrases are Latin
# characters, so bundling a Hebrew/Greek/Cyrillic/Arabic layout would make the
# correct passphrase untypeable and lock the user out.
if [[ -f /etc/vconsole.conf ]]; then
  case $(. /etc/vconsole.conf && echo "${XKBLAYOUT%%,*}") in
    af | am | ara | bd | bg | by | et | ge | gr | il | in | iq | ir | kg | kh | kz | la | lk | mk | mm | mn | mv | np | rs | ru | sy | th | tj | ua) ;;
    *) FILES+=(/etc/vconsole.conf) ;;
  esac
fi
